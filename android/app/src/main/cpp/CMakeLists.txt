cmake_minimum_required(VERSION 3.22.1)
project(RiMS2000RAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS rims_compile=2)
set(CMAKE_JOB_POOL_COMPILE rims_compile)
get_filename_component(RIMS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${RIMS_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/engine" EXCLUDE_FROM_ALL)
target_sources(rims2000r_native PRIVATE "${CMAKE_CURRENT_LIST_DIR}/rims_android.cpp")
target_include_directories(rims2000r_native PRIVATE "${RIMS_ROOT}/source" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(rims2000r_native PRIVATE aaudio android log)
target_link_options(rims2000r_native PRIVATE "-Wl,-z,max-page-size=16384")
